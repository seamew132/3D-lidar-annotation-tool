  function translateString(str) {
    if (!str || typeof str !== 'string') return str;
    let res = str;
    for (let i = 0; i < TRANSLATION_MAP.length; i++) {
      const [ko, en] = TRANSLATION_MAP[i];
      if (res === ko) return en;
      if (res.includes(ko)) res = res.replaceAll(ko, en);
    }
    return res;
  }

  function translateNode(node) {
    if (!node) return;
    if (node.nodeType === Node.ELEMENT_NODE) {
      if (node.id === 'ado-custom-taskbar' || (node.closest && node.closest('#ado-custom-taskbar')) ||
          node.id === 'ado-class-flyout' || (node.closest && node.closest('#ado-class-flyout')) ||
          node.id === 'ado-hotkey-modal-overlay' || (node.closest && node.closest('#ado-hotkey-modal-overlay')) ||
          node.id === 'ado-top-active-class' || (node.closest && node.closest('#ado-top-active-class')) ||
          node.id === 'ado-copy-confirm-modal' || (node.closest && node.closest('#ado-copy-confirm-modal')) ||
          node.id === 'ado-l-confirm-modal' || (node.closest && node.closest('#ado-l-confirm-modal')) ||
          node.id === 'ado-quick-to-modal' || (node.closest && node.closest('#ado-quick-to-modal')) ||
          node.id === 'ado-subclass-desc-overlay' || (node.closest && node.closest('#ado-subclass-desc-overlay')) ||
          (node.classList && node.classList.contains('ado-cam-toggle-btn')) || (node.closest && node.closest('.ado-cam-toggle-btn'))) {
        return;
      }
      const title = node.getAttribute('title');
      if (title && /[\uac00-\ud7af]/.test(title)) node.setAttribute('title', translateString(title));
      const ariaLabel = node.getAttribute('aria-label');
      if (ariaLabel && /[\uac00-\ud7af]/.test(ariaLabel)) node.setAttribute('aria-label', translateString(ariaLabel));
      const placeholder = node.getAttribute('placeholder');
      if (placeholder && /[\uac00-\ud7af]/.test(placeholder)) node.setAttribute('placeholder', translateString(placeholder));
    }

    if (node.nodeType === Node.TEXT_NODE) {
      const val = node.nodeValue;
      if (val && /[\uac00-\ud7af]/.test(val)) {
        const translated = translateString(val);
        if (translated !== val) node.nodeValue = translated;
      }
      return;
    }

    let child = node.firstChild;
    while (child) {
      translateNode(child);
      child = child.nextSibling;
    }
  }

  let isTranslating = false;
  function triggerTranslation() {
    if (isTranslating) return;
    isTranslating = true;
    requestAnimationFrame(() => {
      translateNode(document.body);
      isTranslating = false;
    });
  }

  function findCategoryAndSubclassSelects() {
    const allSelects = Array.from(document.querySelectorAll('select'));

    // Method 1: Find the select whose options have VEHICLE and PARKING_OBSTACLE
    for (const sel of allSelects) {
      const vals = Array.from(sel.options).map(o => o.value);
      if (vals.includes('VEHICLE') && vals.includes('PARKING_OBSTACLE')) {
        const parent = sel.parentElement;
        if (parent) {
          const sibling = parent.querySelector('select:nth-of-type(2)') ||
                          Array.from(parent.querySelectorAll('select')).find(s => s !== sel);
          if (sibling) {
            return { categorySelect: sel, subclassSelect: sibling };
          }
        }
      }
    }

    // Method 2: Match by title (Category / 카테고리)
    const catSel = document.querySelector('select[title="Category"], select[title="카테고리"], select[title*="Category"], select[title*="카테고리"]');
    if (catSel && catSel.parentElement) {
      const subSel = catSel.parentElement.querySelector('select:nth-of-type(2)');
      if (subSel) return { categorySelect: catSel, subclassSelect: subSel };
    }

    return null;
  }

  // Set real React state on a <select> element
  function setSelectRealReactValue(selectEl, targetValue) {
    if (!selectEl) return false;

    // 1. Find matching option (case-insensitive)
    const options = Array.from(selectEl.options);
    const matchedOpt = options.find(o =>
      o.value.toLowerCase() === targetValue.toLowerCase() ||
      o.text.toLowerCase() === targetValue.toLowerCase()
    );

    if (!matchedOpt) {
      console.warn(`[ADO Helper] Option '${targetValue}' not found in:`, selectEl);
      return false;
    }

    const realValue = matchedOpt.value;

    // 2. Select option in DOM
    for (let i = 0; i < selectEl.options.length; i++) {
      selectEl.options[i].selected = (selectEl.options[i].value === realValue);
    }
    selectEl.selectedIndex = matchedOpt.index;

    // 3. Clear React value tracker so React doesn't ignore the change
    if (selectEl._valueTracker) {
      selectEl._valueTracker.setValue('__REACT_FORCE_UPDATE__');
    }

    // 4. Use HTMLSelectElement prototype setter
    const proto = window.HTMLSelectElement.prototype;
    const descriptor = Object.getOwnPropertyDescriptor(proto, 'value');
    if (descriptor && descriptor.set) {
      descriptor.set.call(selectEl, realValue);
    } else {
      selectEl.value = realValue;
    }

    // 5. Dispatch native events with bubbles & composed
    selectEl.dispatchEvent(new Event('input', { bubbles: true, composed: true }));
    selectEl.dispatchEvent(new Event('change', { bubbles: true, composed: true }));

    // 6. Invoke React __reactProps$ onChange directly
    const propsKey = Object.keys(selectEl).find(k => k.startsWith('__reactProps$'));
    if (propsKey && selectEl[propsKey]) {
      const props = selectEl[propsKey];
      if (typeof props.onChange === 'function') {
        try {
          props.onChange({
            target: selectEl,
            currentTarget: selectEl,
            bubbles: true,
            defaultPrevented: false,
            nativeEvent: new Event('change')
          });
          console.log(`[ADO Helper] Called __reactProps onChange directly: ${realValue}`);
        } catch (e) {
          console.warn('[ADO Helper] Direct __reactProps onChange error:', e);
        }
      }
    }

    // 7. Invoke React Fiber onChange directly
    const fiberKey = Object.keys(selectEl).find(k => k.startsWith('__reactFiber$') || k.startsWith('__reactInternalInstance$'));
    if (fiberKey && selectEl[fiberKey]) {
      let cur = selectEl[fiberKey];
      let depth = 0;
      while (cur && depth < 15) {
        if (cur.memoizedProps && typeof cur.memoizedProps.onChange === 'function') {
          try {
            cur.memoizedProps.onChange({
              target: selectEl,
              currentTarget: selectEl,
              bubbles: true,
              defaultPrevented: false,
              nativeEvent: new Event('change')
            });
            console.log(`[ADO Helper] Called React fiber onChange directly: ${realValue}`);
          } catch (e) {}
          break;
        }
        cur = cur.return;
        depth++;
      }
    }

    return true;
  }

  // 5. DIRECT HOTKEY DISPATCH (KEYS 1-9, 0) FOR REFERENCE DIMENSIONS
  // =========================================================================

  function dispatchDimensionHotkey(digitChar) {
    if (!digitChar) return;

    // 1. Ensure focus is NOT inside any select or input
    if (document.activeElement && (document.activeElement.tagName === 'SELECT' || document.activeElement.tagName === 'INPUT')) {
      document.activeElement.blur();
    }

    // 2. Focus 3D canvas so the tool's 3D key listener catches it
    const canvas = document.querySelector('canvas');
    if (canvas && typeof canvas.focus === 'function') {
      canvas.focus();
    }

    const keyCode = 48 + parseInt(digitChar, 10);
    const code = digitChar === '0' ? 'Digit0' : `Digit${digitChar}`;
    const numpadCode = digitChar === '0' ? 'Numpad0' : `Numpad${digitChar}`;
    const numpadKeyCode = 96 + parseInt(digitChar, 10);

    const targets = [window, document, document.body, canvas].filter(Boolean);

    // Standard Digit key (e.g. Digit3)
    const digitOpts = {
      key: digitChar,
      code: code,
      keyCode: keyCode,
      which: keyCode,
      charCode: keyCode,
      bubbles: true,
      cancelable: true,
      composed: true,
      ctrlKey: false,
      altKey: false,
      shiftKey: false
    };

    targets.forEach(t => {
      t.dispatchEvent(new KeyboardEvent('keydown', digitOpts));
      t.dispatchEvent(new KeyboardEvent('keypress', digitOpts));
      t.dispatchEvent(new KeyboardEvent('keyup', digitOpts));
    });

    // Numpad key backup
    const numpadOpts = {
      key: digitChar,
      code: numpadCode,
      keyCode: numpadKeyCode,
      which: numpadKeyCode,
      charCode: numpadKeyCode,
      bubbles: true,
      cancelable: true,
      composed: true,
      ctrlKey: false,
      altKey: false,
      shiftKey: false
    };

    targets.forEach(t => {
      t.dispatchEvent(new KeyboardEvent('keydown', numpadOpts));
      t.dispatchEvent(new KeyboardEvent('keyup', numpadOpts));
    });

    console.log(`[ADO Helper] Dispatched dimension hotkey '${digitChar}' (code: ${code}, keyCode: ${keyCode})`);
  }

  // Asynchronous change: Set Category first, wait for Subclass to unlock, then set Subclass
  async function changeCategoryAndSubclass(group, sub) {
    let selects = findCategoryAndSubclassSelects();
    if (!selects) {
      showToast("⚠️ Chưa chọn box! Hãy click chọn 1 box trong 3D rồi bấm lại.");
      return false;
    }

    let { categorySelect, subclassSelect } = selects;

    // Step 1: Set Category if it's currently different
    if (categorySelect.value.toUpperCase() !== group.category.toUpperCase()) {
      console.log(`[ADO Helper] Setting Category to: ${group.category}`);
      setSelectRealReactValue(categorySelect, group.category);

      // Wait up to 600ms for React to populate the Subclass select and remove 'disabled'
      let populated = false;
      for (let i = 0; i < 30; i++) {
        await new Promise(r => setTimeout(r, 20));

        // Re-query in case React remounted the DOM elements
        const fresh = findCategoryAndSubclassSelects();
        if (fresh) {
          subclassSelect = fresh.subclassSelect;
          categorySelect = fresh.categorySelect;
        }

        if (subclassSelect && !subclassSelect.disabled && subclassSelect.options.length > 1) {
          const hasOption = Array.from(subclassSelect.options).some(o =>
            o.value.toLowerCase() === sub.name.toLowerCase() ||
            o.text.toLowerCase() === sub.name.toLowerCase()
          );
          if (hasOption) {
            populated = true;
            break;
          }
        }
      }

      if (!populated) {
        console.warn(`[ADO Helper] Subclass option '${sub.name}' not yet populated after wait`);
      }
    }

    // Step 2: Set Subclass
    console.log(`[ADO Helper] Setting Subclass to: ${sub.name}`);
    const ok = setSelectRealReactValue(subclassSelect, sub.name);

    // Step 3: Trigger reference dimension hotkey (Keys 1-9, 0)
    if (sub.key) {
      await new Promise(r => setTimeout(r, 60));
      dispatchDimensionHotkey(sub.key);
      setTimeout(() => dispatchDimensionHotkey(sub.key), 150);
    }

    return ok;
  }

  // Handler when user clicks a Subclass button on toolbar
  async function handleSubclassClick(group, sub) {
    const selects = findCategoryAndSubclassSelects();
    if (!selects) {
      showToast(`⚠️ Chưa chọn box! Hãy click chọn 1 box trong 3D trước rồi bấm lại.`);
      return;
    }

    const ok = await changeCategoryAndSubclass(group, sub);
    if (ok) {
      currentActiveSubclass = {
        category: group.category,
        subclass: sub.name,
        title: sub.title,
        refSize: sub.refSize || 'Default',
        key: sub.key
      };
      updateTopActiveBadge();
      const keyMsg = sub.key ? ` (Đã ấn phím <b>${sub.key}</b>)` : '';
      showToast(`✅ Đã đặt: <b>${sub.title}</b> [${sub.refSize || 'OK'}]${keyMsg}`);
    } else {
      showToast(`⚠️ Không thể đổi sang <b>${sub.title}</b>! Kiểm tra lại box đang chọn.`);
    }
  }

  function renderHotkeyContent(query = '') {
    const container = document.getElementById('ado-hotkey-content-area');
    if (!container) return;

    container.innerHTML = '';
    const q = query.toLowerCase();
    let totalMatches = 0;

    HOTKEY_DATA.forEach(group => {
      const filteredItems = group.items.filter(item => {
        if (!q) return true;
        const inKeys = item.keys.some(k => k.toLowerCase().includes(q));
        const inName = item.name.toLowerCase().includes(q);
        const inDesc = item.desc.toLowerCase().includes(q);
        const inGroup = group.group.toLowerCase().includes(q);
        return inKeys || inName || inDesc || inGroup;
      });

      if (filteredItems.length === 0) return;
      totalMatches += filteredItems.length;

      const groupSection = document.createElement('div');
      groupSection.style.display = 'flex';
      groupSection.style.flexDirection = 'column';
      groupSection.style.gap = '8px';

      const groupTitle = document.createElement('div');
      Object.assign(groupTitle.style, {
        fontSize: '13px',
        fontWeight: '700',
        color: group.color || '#38bdf8',
        borderBottom: `1px solid ${group.color}40`,
        paddingBottom: '4px',
        textTransform: 'uppercase',
        letterSpacing: '0.03em'
      });
      groupTitle.textContent = group.group;
      groupSection.appendChild(groupTitle);

      const itemsList = document.createElement('div');
      Object.assign(itemsList.style, {
        display: 'flex',
        flexDirection: 'column',
        gap: '6px'
      });

      filteredItems.forEach(item => {
        const row = document.createElement('div');
        Object.assign(row.style, {
          display: 'flex',
          alignItems: 'flex-start',
          justifyContent: 'space-between',
          gap: '12px',
          padding: '8px 12px',
          borderRadius: '8px',
          backgroundColor: 'rgba(30, 41, 59, 0.4)',
          border: '1px solid rgba(51, 65, 85, 0.4)'
        });

        // Left info
        const infoDiv = document.createElement('div');
        infoDiv.style.flex = '1';
        infoDiv.innerHTML = `
          <div style="font-size:12px; font-weight:600; color:#f1f5f9;">${item.name}</div>
          <div style="font-size:11px; color:#94a3b8; margin-top:2px; line-height:1.4;">${item.desc}</div>
        `;
        row.appendChild(infoDiv);

        // Right keys
        const keysDiv = document.createElement('div');
        Object.assign(keysDiv.style, {
          display: 'flex',
          alignItems: 'center',
          gap: '4px',
          flexShrink: '0'
        });

        item.keys.forEach(k => {
          const kbd = document.createElement('kbd');
          kbd.textContent = k;
          Object.assign(kbd.style, {
            display: 'inline-block',
            padding: '3px 7px',
            fontSize: '11px',
            fontFamily: 'ui-monospace, monospace',
            fontWeight: 'bold',
            color: '#fbbf24',
            backgroundColor: '#1e293b',
            border: '1px solid #475569',
            borderRadius: '5px',
            boxShadow: '0 1px 2px rgba(0,0,0,0.5)',
            whiteSpace: 'nowrap'
          });
          keysDiv.appendChild(kbd);
        });

        row.appendChild(keysDiv);
        itemsList.appendChild(row);
      });

      groupSection.appendChild(itemsList);
      container.appendChild(groupSection);
    });

  // =========================================================================
  // 8. KEYBOARD EVENT INTERCEPTOR: LOCK WEB HOTKEYS WHILE MODAL IS OPEN
  // =========================================================================

  function handleModalKeyEvent(e) {
    if (!isHotkeyModalOpen) return;

    // Esc closes modal
    if (e.key === 'Escape') {
      closeHotkeyModal();
      e.stopPropagation();
      e.preventDefault();
      return;
    }

    // If the event is inside our modal (e.g. typing in search input),
    // let the input receive the keystroke, but stop it from bubbling to the web app Three.js listeners!
    if (e.target && (e.target.id === 'ado-hotkey-search-box' || (e.target.closest && e.target.closest('#ado-hotkey-modal-card')))) {
      // Do not prevent default so text is entered in input, but stop propagation
      e.stopPropagation();
      return;
    }

    // If outside, block completely
    e.stopPropagation();
    e.preventDefault();
  }

  window.addEventListener('keydown', handleModalKeyEvent, true);
  window.addEventListener('keyup', (e) => {
    if (isHotkeyModalOpen) {
      e.stopPropagation();
    }
  }, true);
  window.addEventListener('keypress', (e) => {
    if (isHotkeyModalOpen && e.target && e.target.id !== 'ado-hotkey-search-box') {
      e.stopPropagation();
    }
  }, true);

  function findHeightZFilterButtons() {
    let allSpans = Array.from(document.querySelectorAll('span, div'));
    let zSpan = allSpans.find(el =>
      el.children.length === 0 && (el.textContent.includes('(Z)') || el.textContent.includes('높이(Z)') || el.textContent.includes('Height (Z)'))
    );

    if (!zSpan) {
      ensureSettingsTabOpen();
      allSpans = Array.from(document.querySelectorAll('span, div'));
      zSpan = allSpans.find(el =>
        el.children.length === 0 && (el.textContent.includes('(Z)') || el.textContent.includes('높이(Z)') || el.textContent.includes('Height (Z)'))
      );
    }

    if (zSpan) {
      const row = zSpan.closest('.flex-row') || zSpan.parentElement?.parentElement;
      if (row) {
        const buttons = Array.from(row.querySelectorAll('button'));
        const onBtn = buttons.find(b => b.textContent.trim() === 'ON');
        const offBtn = buttons.find(b => b.textContent.trim() === 'OFF');
        if (onBtn && offBtn) {
          const isOn = onBtn.classList.contains('bg-emerald-500/20') || onBtn.getAttribute('aria-pressed') === 'true';
          return { row, onBtn, offBtn, isOn };
        }
      }
    }
    return null;
  }

  let hasAutoEnabledHeightZ = false;

  function tryAutoEnableHeightZFilter() {
    if (hasAutoEnabledHeightZ) return true;
    const ctrl = findHeightZFilterButtons();
    if (ctrl) {
      if (!ctrl.isOn && ctrl.onBtn) {
        ctrl.onBtn.click();
        hasAutoEnabledHeightZ = true;
        showToast("🌲 Đã tự động BẬT Height(Z) Filter (Lọc tán lá)!");
        return true;
      } else if (ctrl.isOn) {
        hasAutoEnabledHeightZ = true;
        return true;
      }
    }
    return false;
  }

  // Smooth dragging with viewport bounds
  function setupDraggable(element, handle) {
    let isDragging = false;
    let startX = 0, startY = 0;
    let initialLeft = 0, initialTop = 0;

    handle.addEventListener('mousedown', (e) => {
      e.preventDefault();
      e.stopPropagation();

      isDragging = true;
      handle.style.cursor = 'grabbing';

      const rect = element.getBoundingClientRect();
      element.style.bottom = 'auto';
      element.style.transform = 'none';
      element.style.left = `${rect.left}px`;
      element.style.top = `${rect.top}px`;

      startX = e.clientX;
      startY = e.clientY;
      initialLeft = rect.left;
      initialTop = rect.top;

      function onMouseMove(moveEvent) {
        if (!isDragging) return;
        moveEvent.preventDefault();

        const deltaX = moveEvent.clientX - startX;
        const deltaY = moveEvent.clientY - startY;

        let newLeft = initialLeft + deltaX;
        let newTop = initialTop + deltaY;

        const winWidth = window.innerWidth;
        const winHeight = window.innerHeight;
        const elemWidth = element.offsetWidth;
        const elemHeight = element.offsetHeight;

        newLeft = Math.max(10, Math.min(newLeft, winWidth - elemWidth - 10));
        newTop = Math.max(10, Math.min(newTop, winHeight - elemHeight - 10));

        element.style.left = `${newLeft}px`;
        element.style.top = `${newTop}px`;
      }

      function onMouseUp() {
        if (!isDragging) return;
        isDragging = false;
        handle.style.cursor = 'grab';
        window.removeEventListener('mousemove', onMouseMove);
        window.removeEventListener('mouseup', onMouseUp);
      }

      window.addEventListener('mousemove', onMouseMove);
      window.addEventListener('mouseup', onMouseUp);
    });
  }

  function setRealReactInputValue(inputEl, val) {
    if (!inputEl) return false;
    const strVal = val === null || val === undefined ? '' : String(val);

    if (inputEl._valueTracker) {
      inputEl._valueTracker.setValue('__REACT_FORCE_UPDATE__');
    }

    const proto = window.HTMLInputElement.prototype;
    const descriptor = Object.getOwnPropertyDescriptor(proto, 'value');
    if (descriptor && descriptor.set) {
      descriptor.set.call(inputEl, strVal);
    } else {
      inputEl.value = strVal;
    }

    inputEl.dispatchEvent(new Event('input', { bubbles: true, composed: true }));
    inputEl.dispatchEvent(new Event('change', { bubbles: true, composed: true }));
    inputEl.dispatchEvent(new Event('blur', { bubbles: true, composed: true }));

    const propsKey = Object.keys(inputEl).find(k => k.startsWith('__reactProps$'));
    if (propsKey && inputEl[propsKey]) {
      if (typeof inputEl[propsKey].onChange === 'function') {
        try {
          inputEl[propsKey].onChange({
            target: inputEl,
            currentTarget: inputEl,
            bubbles: true,
            defaultPrevented: false
          });
        } catch (e) {}
      }
      if (typeof inputEl[propsKey].onBlur === 'function') {
        try {
          inputEl[propsKey].onBlur({
            target: inputEl,
            currentTarget: inputEl,
            bubbles: true,
            defaultPrevented: false
          });
        } catch (e) {}
      }
    }

    const fiberKey = Object.keys(inputEl).find(k => k.startsWith('__reactFiber$'));
    if (fiberKey && inputEl[fiberKey]) {
      let cur = inputEl[fiberKey];
      let depth = 0;
      while (cur && depth < 15) {
        if (cur.memoizedProps && typeof cur.memoizedProps.onChange === 'function') {
          try {
            cur.memoizedProps.onChange({ target: inputEl, currentTarget: inputEl });
          } catch (e) {}
          break;
        }
        cur = cur.return;
        depth++;
      }
    }

    return true;
  }

  function writeCameraAttrs(data) {
    if (!data) return false;
    const inputs = getCameraAttrInputs();
    if (!inputs) return false;

    let count = 0;
    for (const cam of ['front', 'rear', 'left', 'right']) {
      if (data[cam] && inputs[cam]) {
        if (data[cam].t !== undefined && data[cam].t !== '') {
          setRealReactInputValue(inputs[cam].tInput, data[cam].t);
          count++;
        }
        if (data[cam].o !== undefined && data[cam].o !== '') {
          setRealReactInputValue(inputs[cam].oInput, data[cam].o);
          count++;
        }
      }
    }
    return count > 0;
  }

  function findCurrentActiveBoxId() {

  // Lấy các phần tử DOM của từng camera
  function getCameraElements(camName) {
    const img = document.querySelector(`img[alt="${camName}"]`);
    if (!img) return null;
    const container = img.closest('.relative') || img.parentElement;
    const svg = container ? container.querySelector('svg') : null;

  // Tự động nhận diện DUY NHẤT box ĐANG ĐƯỢC CHỌN xuất hiện trên camera nào
  function autoDetectCamerasForBox() {
    const detected = new Set();
    const curBoxId = findCurrentActiveBoxId();

    CAM_NAMES.forEach(camName => {
      const cam = getCameraElements(camName);
      if (!cam || !cam.svg) return;

      let hasSelectedBox = false;

      // 1. Quét theo thẻ <text> trong SVG: Khung nhìn của camera nào chứa box đang chọn
      // sẽ có thẻ text in đúng số tracking ID của box đó (ví dụ: <text>1</text>)
      if (curBoxId) {
        const textNodes = Array.from(cam.svg.querySelectorAll('text'));
        const matched = textNodes.some(t => (t.textContent || '').trim() === String(curBoxId));
        if (matched) {
          hasSelectedBox = true;
        }
      }

      // 2. Quét theo màu vàng (#ffff00): Box đang chọn trên 2D luôn có nét viền (stroke) hoặc chữ (fill) màu vàng thuần
      if (!hasSelectedBox) {
        const yellowNodes = cam.svg.querySelectorAll('[stroke="#ffff00"], [stroke*="255, 255, 0"], [fill="#ffff00"], [fill*="255, 255, 0"]');
        if (yellowNodes.length > 0) {
          hasSelectedBox = true;
        }
      }

      if (hasSelectedBox) {
        detected.add(camName);
      }
    });

    return detected;
  }

  // Kiểm tra xem box có nằm sát rìa ảnh hoặc sát rìa lens mắt cá hay không
  function isBoxNearCameraEdge(camName, curBoxId) {
    const cam = getCameraElements(camName);
    if (!cam || !cam.svg) return false;

    let targetPolylines = [];
    if (curBoxId) {
      const textEls = Array.from(cam.svg.querySelectorAll('text'));
      const matchedText = textEls.find(t => (t.textContent || '').trim() === String(curBoxId));
      if (matchedText) {
        const g = matchedText.closest('g');
        if (g) targetPolylines = Array.from(g.querySelectorAll('polyline'));
      }
    }

    if (targetPolylines.length === 0) {
      targetPolylines = Array.from(cam.svg.querySelectorAll('polyline[stroke="#ffff00"]'));
    }

    if (targetPolylines.length === 0) return false;

    const centerX = 912;
    const centerY = 768;
    const lensRadiusThreshold = 680; // Khoảng cách tới tâm > 680 là sát rìa lens tròn

    for (const poly of targetPolylines) {
      const ptsAttr = poly.getAttribute('points');
      if (!ptsAttr) continue;
      const coords = ptsAttr.trim().split(/[\s,]+/);
      for (let i = 0; i < coords.length - 1; i += 2) {
        const x = parseFloat(coords[i]);
        const y = parseFloat(coords[i + 1]);
        if (isNaN(x) || isNaN(y)) continue;

        // 1. Sát rìa khung ảnh chữ nhật (cách viền < 50px hoặc vượt ra ngoài)
        if (x <= 50 || x >= 1774 || y <= 50 || y >= 1486) {
          return true;
        }

        // 2. Sát rìa vòng tròn ống kính mắt cá fisheye
        const dist = Math.sqrt((x - centerX) * (x - centerX) + (y - centerY) * (y - centerY));
        if (dist >= lensRadiusThreshold) {
          return true;
        }
      }
    }

    return false;
  }

  // Kiểm tra xem camera có đang bị ẩn box 2D (\uC228\uAE40) hay không
  function isCamForciblyHidden(camName) {
    const camLower = camName.toLowerCase();
    const allLis = document.querySelectorAll('li');
    for (const li of allLis) {
      const textSpans = Array.from(li.querySelectorAll('span')).map(s => s.textContent.trim().toLowerCase());
      if (textSpans.includes(camLower)) {
        const hasBadge = Array.from(li.querySelectorAll('span')).some(s => s.textContent.trim() === '\uC228\uAE40' || s.textContent.trim().toLowerCase() === 'hidden');
        const hasReprojectBtn = Array.from(li.querySelectorAll('button')).some(b => b.textContent.includes('\uC7AC\uD22C\uC601') || b.title?.includes('\uC7AC\uD22C\uC601'));
        if (hasBadge || hasReprojectBtn) return true;
      }
    }
    return false;
  }

  // Khôi phục lại box 2D từ 3D Cuboid (Reprojection [\uC7AC\uD22C\uC601])
  async function reprojectCameraBox(camName) {
    const camLower = camName.toLowerCase();
    let clicked = false;
    // Cách 1: Bấm trực tiếp nút [\uC7AC\uD22C\uC601] ở Camera 속성 bên sidebar
    const allLis = document.querySelectorAll('li');
    for (const li of allLis) {
      const textSpans = Array.from(li.querySelectorAll('span')).map(s => s.textContent.trim().toLowerCase());
      if (textSpans.includes(camLower)) {
        const btn = Array.from(li.querySelectorAll('button')).find(b => b.textContent.includes('\uC7AC\uD22C\uC601') || b.title?.includes('\uC7AC\uD22C\uC601'));
        if (btn) {
          const origConfirm = window.confirm;
          try {
            window.confirm = () => true;
            btn.click();
            clicked = true;
          } finally {
            window.confirm = origConfirm;
          }
          break;
        }
      }
    }
    if (clicked) {
      await new Promise(r => setTimeout(r, 60));
      return true;
    }

    // Cách 2: Vào 2D box mode và gửi phím R
    const cam = getCameraElements(camName);
    if (cam && cam.container) {
      cam.container.dispatchEvent(new MouseEvent('contextmenu', { bubbles: true, cancelable: true, view: window }));
      await new Promise(r => setTimeout(r, 40));
      window.dispatchEvent(new KeyboardEvent('keydown', { code: 'KeyR', key: 'r', keyCode: 82, which: 82, bubbles: true, cancelable: true }));
      await new Promise(r => setTimeout(r, 40));
      cam.container.dispatchEvent(new MouseEvent('contextmenu', { bubbles: true, cancelable: true, view: window }));
      await new Promise(r => setTimeout(r, 40));
      return true;
    }
    return false;
  }

  // Xóa box 2D cưỡng bức (Force Hide [\uC228\uAE40])
  async function forceHideCameraBox(camName) {
    const cam = getCameraElements(camName);
    if (!cam || !cam.container) return false;

    // Vào chế độ 2D Box của camera này
    cam.container.dispatchEvent(new MouseEvent('contextmenu', { bubbles: true, cancelable: true, view: window }));
    await new Promise(r => setTimeout(r, 40));

    // Gửi phím B để xóa box 2D (l5 forceHide)
    window.dispatchEvent(new KeyboardEvent('keydown', { code: 'KeyB', key: 'b', keyCode: 66, which: 66, bubbles: true, cancelable: true }));
    await new Promise(r => setTimeout(r, 40));

    // Thoát chế độ 2D Box
    cam.container.dispatchEvent(new MouseEvent('contextmenu', { bubbles: true, cancelable: true, view: window }));
    await new Promise(r => setTimeout(r, 40));

    // Gán T=0 và O=0 cho camera này để đúng chuẩn gán nhãn khi xóa box 2D
    const camKey = camName.toLowerCase();
    syncQuickTOValue(camKey, 't', '0');
    syncQuickTOValue(camKey, 'o', '0');

    return true;
  }

  // Network hook to cache frame data from fetch
  try {
    const origFetch = window.fetch;
    window.fetch = async function (...args) {
      const res = await origFetch.apply(this, args);
      try {
        const clone = res.clone();
        clone.json().then(data => {
          const scan = (obj) => {
            if (!obj) return;
            if (Array.isArray(obj)) {
              obj.forEach(scan);
            } else if (typeof obj === 'object') {
              const id = obj.id ?? obj.tracking_id ?? obj.object_id ?? obj.obj_id;
              const props = obj.camera_properties ?? obj.camera_attributes ?? obj.cameraAttributes ?? obj.cameraProps;
              if (id !== undefined && props && typeof props === 'object') {
                const map = {};
                for (const cam of ['front', 'rear', 'left', 'right']) {
                  if (props[cam]) {
                    const t = props[cam].truncation ?? props[cam].t;
                    const o = props[cam].occlusion ?? props[cam].o;
                    map[cam] = {
                      t: t !== undefined && t !== null ? String(t) : '',
                      o: o !== undefined && o !== null ? String(o) : ''
                    };
                  }
                }
                if (Object.keys(map).length > 0) {
                  camAttrCacheById.set(String(id), map);
                  try {
                    let stored = JSON.parse(localStorage.getItem('ado_temp_cam_attrs') || '{}');
                    stored[String(id)] = map;
                    localStorage.setItem('ado_temp_cam_attrs', JSON.stringify(stored));
                  } catch (err) {}
                }
              }
              if (Array.isArray(obj.annotations)) obj.annotations.forEach(scan);
              if (Array.isArray(obj.objects)) obj.objects.forEach(scan);
              if (Array.isArray(obj.cuboids)) obj.cuboids.forEach(scan);
              if (Array.isArray(obj.boxes)) obj.boxes.forEach(scan);
            }
          };
          scan(data);
        }).catch(() => {});
      } catch (e) {}
      return res;
    };
  } catch (e) {}

  function downloadJson(data, filename = 'ado_saved_cam_attrs.json') {
    try {
      const jsonStr = JSON.stringify(data, null, 2);
      const blob = new Blob([jsonStr], { type: 'application/json' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = filename;
      document.body.appendChild(a);
      a.click();
      setTimeout(() => {
        if (a.parentNode) a.parentNode.removeChild(a);
        URL.revokeObjectURL(url);
      }, 200);
    } catch (e) {
      console.error('Error downloading JSON:', e);
    }
  }

  function getAllObjectButtons() {
    const aside = document.querySelector('aside') || document.body;
    const items = [];
    const seenIds = new Set();

    // 1. Tìm tất cả các button dòng đối tượng có chứa số ID font-mono (chuẩn cấu trúc ALCHERA ADO)
    const allButtons = Array.from(aside.querySelectorAll('button'));
    for (const btn of allButtons) {
      const mono = btn.querySelector('.font-mono');
      if (mono && /^\d+$/.test(mono.textContent.trim())) {
        const id = mono.textContent.trim();
        if (!seenIds.has(id)) {
          seenIds.add(id);
          items.push({ id, element: btn, button: btn });
        }
      }
    }

    // 2. Dự phòng: Quét theo các Category
    if (items.length === 0) {
      const CLASS_NAMES = ['VEHICLE', 'Unspecified', 'PEDESTRIAN', 'CYCLIST', 'STATIC', 'TRAFFIC_CONE', 'BARRIER'];
      const candidates = Array.from(aside.querySelectorAll('div, button, li')).filter(el => {
        if (el.children.length > 8 || el.children.length === 0) return false;
        const t = el.textContent || '';
        if (t.includes('Object (') || t.includes('객체 (')) return false;
        return CLASS_NAMES.some(c => t.includes(c)) && /\b\d+\b/.test(t);
      });
      for (const el of candidates) {
        const txt = el.textContent.trim();
        const m = txt.match(/^(\d+)/) || txt.match(/\b(\d+)\s+(Unspecified|VEHICLE|PEDESTRIAN|CYCLIST|STATIC)/i);
        if (m && !seenIds.has(m[1])) {
          let row = el.closest('button') || el.closest('.cursor-pointer') || el;
          seenIds.add(m[1]);
          items.push({ id: m[1], element: row, button: row });
        }
      }
    }

    return items;
  }

  async function selectBoxItem(item) {

    } catch (e) {}

    // Click đơn vào dòng để chuyển sang box này
    item.element.click();

  function extractDirectFromReact() {
    const allSpans = Array.from(document.querySelectorAll('span'));
    const objHeader = allSpans.find(s => s.textContent && (s.textContent.includes('Object (') || s.textContent.includes('객체 (')));
    const startEl = objHeader ? (objHeader.closest('button') || objHeader) : document.querySelector('aside') || document.body;

    const fiberKey = Object.keys(startEl).find(k => k.startsWith('__reactFiber$'));
    if (!fiberKey) return null;

    let fiber = startEl[fiberKey];
    let depth = 0;
    while (fiber && depth < 40) {
      if (fiber.memoizedProps) {
        for (const [k, v] of Object.entries(fiber.memoizedProps)) {
          if (Array.isArray(v) && v.length > 0 && typeof v[0] === 'object' && v[0] !== null) {
            if (v[0].id !== undefined || v[0].tracking_id !== undefined || v[0].camera_properties || v[0].cameraAttributes) {
              return v;
            }
          }
        }
      }
      let s = fiber.memoizedState;
      while (s) {
        if (s.memoizedState) {
          const val = s.memoizedState;
          if (Array.isArray(val) && val.length > 0 && typeof val[0] === 'object' && val[0] !== null) {
            if (val[0].id !== undefined || val[0].tracking_id !== undefined || val[0].camera_properties || val[0].cameraAttributes) {
              return val;
            }
          }
        }
        s = s.next;
      }
      fiber = fiber.return;
      depth++;
    }
    return null;
  }

  async function saveTempCamAttrs() {
    const objItems = getObjectListItems();
    console.log('[ADO] Save - Scanning Object List rows (sorted top to bottom):', objItems.map(o => o.id));

    const passData = { 1: {}, 2: {} };

    if (objItems.length > 0) {
      showToast(`🚀 Bắt đầu quét 2 lượt từ ô đầu tiên (#${objItems[0].id})...`);

      // Quét 2 lượt đối soát từ ô đầu tiên
      for (let pass = 1; pass <= 2; pass++) {
        for (let idx = 0; idx < objItems.length; idx++) {
          const item = objItems[idx];
          await selectBoxItem(item);
          showToast(`⏳ Lượt ${pass}/2: Quét box ${idx + 1}/${objItems.length} (#${item.id})...`);
          await new Promise(r => setTimeout(r, 160));
          let attrs = readCameraAttrs();
          if (!attrs) {
            await new Promise(r => setTimeout(r, 120));
            attrs = readCameraAttrs();
          }
          if (attrs) {
            passData[pass][item.id] = attrs;
          }
        }
      }
    } else {
      // Fallback if object list not found
      recordActiveBoxCamAttrs();
      const canvas = document.querySelector('canvas') || document.body;
      if (canvas) canvas.focus();
      for (let i = 0; i < 16; i++) {
        canvas.dispatchEvent(new KeyboardEvent('keydown', { key: 'Tab', code: 'Tab', keyCode: 9, which: 9, bubbles: true, cancelable: true }));
        await new Promise(r => setTimeout(r, 130));
        recordActiveBoxCamAttrs();
      }
    }

    // Đối soát 2 lượt & Hợp nhất: ƯU TIÊN GIÁ TRỊ CÓ DỮ LIỆU
    const finalData = {};
    const differences = [];
    const allIds = new Set([...Object.keys(passData[1]), ...Object.keys(passData[2])]);

    for (const id of allIds) {
      const p1 = passData[1][id] || {};
      const p2 = passData[2][id] || {};
      finalData[id] = {};

      for (const cam of ['front', 'rear', 'left', 'right']) {
        const t1 = p1[cam]?.t ?? '';
        const t2 = p2[cam]?.t ?? '';
        const o1 = p1[cam]?.o ?? '';
        const o2 = p2[cam]?.o ?? '';

        // Ưu tiên cái nào có số
        const finalT = (t2 !== '' && t2 !== null) ? t2 : t1;
        const finalO = (o2 !== '' && o2 !== null) ? o2 : o1;

        finalData[id][cam] = { t: finalT, o: finalO };

        if ((t1 !== '' && t2 === '') || (t1 === '' && t2 !== '')) {
          differences.push({ id, cam, prop: 'T', chosen: finalT });
        }
        if ((o1 !== '' && o2 === '') || (o1 === '' && o2 !== '')) {
          differences.push({ id, cam, prop: 'O', chosen: finalO });
        }
      }
    }

    // Merge with any cache already in memory
    for (const [id, val] of camAttrCacheById.entries()) {
      if (!finalData[id]) finalData[id] = val;
    }

    const count = Object.keys(finalData).length;
    if (count === 0) {
      showToast("⚠️ Chưa có thông số T/O nào để lưu tạm! Hãy đảm bảo đã nhập T/O cho các xe.");
      return false;
    }

    // Save to memory cache and localStorage
    for (const [id, val] of Object.entries(finalData)) {
      camAttrCacheById.set(id, val);
    }
    try {
      localStorage.setItem('ado_temp_cam_attrs', JSON.stringify(finalData));
      localStorage.setItem('ado_temp_cam_time', String(Date.now()));
    } catch (e) {}

    const idsList = Object.keys(finalData).join(', #');
    if (differences.length === 0) {
      showToast(`💾 Đã lưu tạm ${count} box! Mở bảng kiểm soát...`);
    } else {
      const diffIds = [...new Set(differences.map(d => d.id))].join(', #');
      showToast(`⚡ Đã lưu tạm ${count} box (đối soát khớp #${diffIds})! Mở bảng kiểm soát...`);
    }

    // Mở ngay Bảng kiểm soát trực quan (không tải file json nữa)
    showCamAttrsInspectionModal(finalData, 'Quét Từng Xe');

    return true;
  }

  async function pasteTempCamAttrs() {
    let stored = {};
    try {
      stored = JSON.parse(localStorage.getItem('ado_temp_cam_attrs') || '{}');
    } catch (e) {}

    for (const [id, val] of camAttrCacheById.entries()) {
      if (!stored[id]) stored[id] = val;
    }

    const totalKnown = Object.keys(stored).length;
    if (totalKnown === 0) {
      showToast("⚠️ Bộ nhớ trống! Hãy bấm '💾 Lưu Tạm T/O' ở frame trước trước khi dán.");
      return false;
    }

    const idsList = Object.keys(stored).join(', #');
    showToast(`⏳ Bắt đầu dán T/O cho các box: #${idsList}...`);

    let updatedCount = 0;
    const visitedIds = new Set();

    // 1. Quét tất cả các box trong Object List
    const allBtns = getAllObjectButtons();
    console.log('[ADO] Paste - All Object rows:', allBtns.map(o => o.id));

    if (allBtns.length > 0) {
      for (let idx = 0; idx < allBtns.length; idx++) {
        const item = allBtns[idx];
        if (stored[item.id]) {
          await selectBoxItem(item);
          showToast(`⏳ Đang dán T/O: box ${idx + 1} / ${allBtns.length} (#${item.id})...`);
          await new Promise(r => setTimeout(r, 160));
          let ok = writeCameraAttrs(stored[item.id]);
          if (!ok) {
            await new Promise(r => setTimeout(r, 140));
            ok = writeCameraAttrs(stored[item.id]);
          }
          if (ok) {
            visitedIds.add(item.id);
            updatedCount++;
            await new Promise(r => setTimeout(r, 80));
          }
        }
      }
    }

    // 2. TẬP TRUNG XỬ LÝ NHỮNG BOX CÒN KẸT CHỮ "T/O" (CẢNH BÁO CHƯA NHẬP)
    await new Promise(r => setTimeout(r, 180));
    let stuckBoxes = getUnenteredBoxItems();
    console.log('[ADO] Checking remaining boxes stuck with T/O badge:', stuckBoxes.map(b => b.id));

    if (stuckBoxes.length > 0) {
      for (const item of stuckBoxes) {
        if (stored[item.id]) {
          console.log(`[ADO] 🎯 Đang dán bù cho box #${item.id} còn kẹt chữ T/O...`);
          showToast(`🎯 Đang dán bù cho box #${item.id} (còn kẹt chữ T/O)...`);
          item.button.click();
          await new Promise(r => setTimeout(r, 200));
          const ok = writeCameraAttrs(stored[item.id]);
          if (ok) {
            visitedIds.add(item.id);
            updatedCount++;
            await new Promise(r => setTimeout(r, 100));
          }
        }
      }
    }

    // 3. Fallback: Tab cycling nếu vẫn còn box kẹt chữ T/O
    stuckBoxes = getUnenteredBoxItems();
    if (stuckBoxes.length > 0) {
      const canvas = document.querySelector('canvas') || document.body;
      if (canvas) canvas.focus();

      for (let step = 0; step < 20; step++) {
        stuckBoxes = getUnenteredBoxItems();
        if (stuckBoxes.length === 0) break;

        const curId = findCurrentActiveBoxId();
        if (curId && stored[curId] && stuckBoxes.some(b => b.id === curId)) {
          const ok = writeCameraAttrs(stored[curId]);
          if (ok) {
            visitedIds.add(curId);
            updatedCount++;
            await new Promise(r => setTimeout(r, 90));
          }
        }

        canvas.dispatchEvent(new KeyboardEvent('keydown', { key: 'Tab', code: 'Tab', keyCode: 9, which: 9, bubbles: true, cancelable: true }));
        await new Promise(r => setTimeout(r, 120));
      }
    }

    // 4. Báo cáo kết quả
    const remainingStuck = getUnenteredBoxItems();
    if (remainingStuck.length === 0) {
      showToast(`🎉 Tuyệt vời! Đã dán hoàn tất và XÓA SẠCH toàn bộ chữ T/O trên tất cả các box!`);
    } else {
      const remList = remainingStuck.map(r => r.id).join(', #');
      showToast(`⚠️ Còn box (#${remList}) kẹt chữ T/O, bạn chỉ cần bấm '📥 Dán T/O' thêm lần nữa để dán nốt!`);
    }

    console.log('[ADO] Pasted to boxes:', Array.from(visitedIds));

    // Sau khi dán xong: Mở ngay Bảng kiểm soát T/O (có nút Check và nút Xóa lưu tạm)
    if (updatedCount > 0) {
      setTimeout(() => {
        showCamAttrsInspectionModal(stored, 'Sau Khi Dán (Paste)');
      }, 500);
    }

    return updatedCount > 0;
  }

  // Intercept Ctrl + C, Ctrl + V, and L
  window.addEventListener('keydown', (e) => {
    const active = document.activeElement;
    if (active && (active.tagName === 'INPUT' || active.tagName === 'TEXTAREA')) {
      if (active.id === 'ado-hotkey-search-box') return;
      if (active.type === 'text' && window.getSelection()?.toString().length > 0) return;
    }

    // 0. Phím ` hoặc ~ (Bật/Tắt Viewmode Camera 2D)
    if (!e.ctrlKey && !e.altKey && !e.metaKey && (e.code === 'Backquote' || e.key === '`' || e.key === '~')) {
      if (active && (active.tagName === 'INPUT' || active.tagName === 'TEXTAREA')) return;
      e.preventDefault();
      e.stopPropagation();
      toggleViewmode();
      return;
    }

    // 0.1 Phím , (Dấu phẩy) để Mở / Đóng Bảng T/O Chỉnh Nhanh (chỉ khi đang ở Viewmode)
    if (!e.ctrlKey && !e.altKey && !e.metaKey && (e.key === ',' || e.code === 'Comma')) {
      if (active && (active.tagName === 'INPUT' || active.tagName === 'TEXTAREA')) return;
      if (isViewmodeActive) {
        e.preventDefault();
        e.stopPropagation();
        toggleQuickTOModal();
        return;
      }
    }

    // 0.2 Phím Escape đóng Bảng T/O Chỉnh Nhanh nếu đang mở
    if (e.key === 'Escape') {
      const qModal = document.getElementById('ado-quick-to-modal');
      if (qModal && qModal.style.display !== 'none') {
        e.preventDefault();
        e.stopPropagation();
        qModal.style.display = 'none';
        return;
      }
    }

    // 1. Phím L (Copy từ Frame trước)
    if (!e.ctrlKey && !e.altKey && !e.metaKey && e.key.toLowerCase() === 'l') {
      if (isBypassingL) return;
      e.preventDefault();
      e.stopPropagation();
      showLConfirmModal();
      return;
    }



    // 2. Ctrl + C (Copy)
    if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'c' && !e.shiftKey && !e.altKey) {
      const camData = readCameraAttrs();
      if (camData) {
        setTimeout(() => showCopyConfirmModal(camData), 50);
      }
    }

    // 3. Ctrl + V or Ctrl + Shift + V (Paste with Class, Subclass & Camera T/O)
    if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'v') {
      if (savedBoxData) {
        let attempts = 0;
        const pasteInterval = setInterval(async () => {
          attempts++;
          let appliedClass = false;
          if (savedBoxData.category && savedBoxData.subclass) {
            const grp = PJT64_CLASSES.find(c => c.category.toUpperCase() === savedBoxData.category.toUpperCase());
            const sub = grp?.subclasses.find(s => s.name.toLowerCase() === savedBoxData.subclass.toLowerCase());
            if (grp && sub) {
              appliedClass = await changeCategoryAndSubclass(grp, sub);
            }
          }
          let okCam = true;
          if (copyIncludesCam && savedBoxData.camAttrs) {
            okCam = writeCameraAttrs(savedBoxData.camAttrs);
          }
          if (appliedClass || okCam || attempts > 12) {
            clearInterval(pasteInterval);
            const classInfo = savedBoxData.category ? `<b>${savedBoxData.category} / ${savedBoxData.subclass}</b> ` : '';
            showToast(`✨ Đã dán hoàn tất: ${classInfo}${copyIncludesCam && savedBoxData.camAttrs ? '+ Camera T/O' : ''}!`);
          }
        }, 140);
      }
    }
  }, true);

  function init() {
    createBottomTaskbar();
    triggerTranslation();
    tryAutoSelectROI();
    tryAutoEnableHeightZFilter();

    const observer = new MutationObserver((mutations) => {
      let shouldTranslate = false;

      for (let i = 0; i < mutations.length; i++) {
        const m = mutations[i];
        if (m.target && (m.target.id === 'ado-custom-taskbar' || m.target.id === 'ado-class-flyout' || m.target.id === 'ado-hotkey-modal-overlay' || m.target.id === 'ado-top-active-class' || m.target.id === 'ado-copy-confirm-modal' || m.target.id === 'ado-l-confirm-modal' || m.target.id === 'ado-quick-to-modal' || m.target.id === 'ado-subclass-desc-overlay' || (m.target.classList && m.target.classList.contains('ado-cam-toggle-btn')))) continue;
        if (m.target && m.target.closest && (m.target.closest('#ado-custom-taskbar') || m.target.closest('#ado-class-flyout') || m.target.closest('#ado-hotkey-modal-overlay') || m.target.closest('#ado-top-active-class') || m.target.closest('#ado-copy-confirm-modal') || m.target.closest('#ado-l-confirm-modal') || m.target.closest('#ado-quick-to-modal') || m.target.closest('#ado-subclass-desc-overlay') || m.target.closest('.ado-cam-toggle-btn'))) continue;

        shouldTranslate = true;
        break;
      }

      if (shouldTranslate) {
        triggerTranslation();
      }

      if (!hasAutoSelectedROI) {
        tryAutoSelectROI();
      }

      if (!hasAutoEnabledHeightZ) {
        tryAutoEnableHeightZFilter();
      }
    });

    observer.observe(document.body, {
      childList: true,
      subtree: true,
      characterData: true
    });

    let retryCount = 0;
    const interval = setInterval(() => {
      retryCount++;
      tryAutoSelectROI();
      tryAutoEnableHeightZFilter();
      if ((hasAutoSelectedROI && hasAutoEnabledHeightZ) || retryCount > 40) {
        clearInterval(interval);
      }
    }, 500);
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
